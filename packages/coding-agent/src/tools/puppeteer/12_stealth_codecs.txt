const parseInput = (arg) => {
	const [mime, codecStr] = arg.trim().split(";");
	let codecs = [];
	if (codecStr && codecStr.includes('codecs="')) {
		codecs = codecStr
			.trim()
			.replace('codecs="', "")
			.replace('"', "")
			.trim()
			.split(",")
			.filter(Boolean)
			.map(item => item.trim());
	}
	return { mime, codecStr, codecs };
};

		}
		const { mime, codecs } = parseInput(String(args[0]));

			return "probably";
		}
		if (mime === "audio/x-m4a" && !codecs.length) {
			return "maybe";
		}
		if (mime === "audio/aac" && !codecs.length) {
			return "probably";
		}
		return nativeResult;
	};

		enumerable: canPlayTypeDescriptor.enumerable,
	});
